5️⃣ main.tf (PRIMARY PIPELINE – FULL)

resource "aws_codepipeline" "primary" {
  name     = "${var.project_prefix}-${var.environment}-Primary-Pipeline"
  role_arn = var.pipeline_role_arn

  artifact_store {
    location = var.artifact_bucket
    type     = "S3"
  }

  #################################
  # SOURCE
  #################################
  stage {
    name = "Source"

    action {
      name             = "Source"
      category         = "Source"
      owner            = "AWS"
      provider         = "CodeCommit"
      version          = "1"
      output_artifacts = ["SourceArtifact"]

      configuration = {
        RepositoryName       = var.source_repo
        BranchName           = var.source_branch
        PollForSourceChanges = false
      }
    }
  }

  #################################
  # VALIDATION
  #################################
  stage {
    name = "Validation"

    action {
      name     = "Json-Validator"
      category = "Build"
      owner    = "AWS"
      provider = "CodeBuild"
      version  = "1"

      input_artifacts  = ["SourceArtifact"]
      output_artifacts = ["ValidatedOutput"]

      configuration = {
        ProjectName = "RM-GN-NON-PROD-Json-Validator"
      }
    }

    action {
      name     = "Metadata-Validator"
      category = "Build"
      owner    = "AWS"
      provider = "CodeBuild"
      version  = "1"

      input_artifacts = ["SourceArtifact", "ValidatedOutput"]

      configuration = {
        ProjectName = "RM-GN-NON-PROD-Metadata-Validator"
      }
    }
  }

  #################################
  # PRE-DEPLOY
  #################################
  stage {
    name = "PreDeploy"

    action {
      name     = "Copy-Artifacts"
      category = "Build"
      owner    = "AWS"
      provider = "CodeBuild"
      version  = "1"

      input_artifacts = ["SourceArtifact", "ValidatedOutput"]

      configuration = {
        ProjectName = "RM-GN-NON-PROD-Copy-Artifacts"
      }
    }

    action {
      name     = "Backup-Execution"
      category = "Build"
      owner    = "AWS"
      provider = "CodeBuild"
      version  = "1"

      input_artifacts = ["SourceArtifact", "ValidatedOutput"]

      configuration = {
        ProjectName = "RM-GN-NON-PROD-Backup-Execution"
      }
    }
  }

  #################################
  # START DEPLOY APPROVAL
  #################################
  stage {
    name = "Approval-To-Deploy"

    action {
      name     = "Manual-Approval"
      category = "Approval"
      owner    = "AWS"
      provider = "Manual"
      version  = "1"
    }
  }

  #################################
  # CONTROLLERS → GATES → APPROVALS
  #################################
  dynamic "stage" {
    for_each = var.controller_gates
    content {
      name = stage.value.controller

      action {
        name     = stage.value.controller
        category = "Build"
        owner    = "AWS"
        provider = "CodeBuild"
        version  = "1"

        input_artifacts = ["ValidatedOutput"]

        configuration = {
          ProjectName = stage.value.controller
        }
      }
    }
  }

  dynamic "stage" {
    for_each = var.controller_gates
    content {
      name = stage.value.gate

      action {
        name     = stage.value.gate
        category = "Build"
        owner    = "AWS"
        provider = "CodeBuild"
        version  = "1"

        input_artifacts = ["ValidatedOutput"]

        configuration = {
          ProjectName = stage.value.gate
        }
      }
    }
  }

  dynamic "stage" {
    for_each = var.controller_gates
    content {
      name = "${stage.value.gate}-Approval"

      action {
        name     = "Manual-Approval"
        category = "Approval"
        owner    = "AWS"
        provider = "Manual"
        version  = "1"
      }
    }
  }
}
